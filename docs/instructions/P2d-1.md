# P2d-1：前端进程架构（第一步：进程模型、写入通道、转发）

本文件给出详细设计与参考代码。参考代码是"骨架"：接口、顺序、边界条件按它来，细节（日志、类型注解、注释）按项目风格补全。与现有代码的衔接点都标出了文件与函数名，不需要重新通读代码库。

## 〇、目标与总体结构

```
                       ┌─ 前端进程 i（i = 0..n-1）───────────────────────────┐
客户端 ─ 8080 端口 ────┤ POST /v1/ingest：读取 → 解压 → 预留 → 解析 → 发给引擎 │──┐ Unix 套接字（写入通道）
（SO_REUSEPORT，       │ 其他所有请求：原样转发给引擎                         │──┼─┐ Unix 套接字（内部 HTTP）
 内核按连接分配）      └──────────────────────────────────────────────────────┘  │ │
                                                                                 ↓ ↓
                       ┌─ 引擎进程（主进程，单写者）──────────────────────────────┐
                       │ 写入通道：读取 BATCH → 去重、写入统计结构 → 回复 ACK     │
                       │ 内部 HTTP：现有 create_app 的全部路由（只在 Unix 套接字上）│
                       │ tick、降级控制；把降级状态写进共享内存                    │
                       └──────────────────────────────────────────────────────────┘
```

依据（docs/lanes/P2-notes.md）：P2c 中 512 MB、20 万条/秒、10 分钟，主进程 CPU 约 95%，可持续约 18.7 万；P2b 剖析中主进程约 12% 在进程池通信、约 12% 在 HTTP、约 15% 在事件循环。前端架构把这三项的大部分移出引擎。

## 规则

- 不修改冻结的基础模块（含 dedup.py、config.py、segtree.py、bounds.py、hashing.py、sketch/）
- 必须保持的语义：在途额度在解析之前预留；去重只在引擎里做；第 4 级在读取请求体之前拒绝；单条不合法只计入 invalid；对外的状态码、错误格式、头部与单进程模式完全一致
- 前端数为 0 时，行为与现在的单进程模式完全相同（包括现有的全部测试不需要修改就能通过）
- 每完成一节就提交一次；决策与结果追加进 docs/lanes/P2-notes.md 的新一节"P2d-1"

def resolve_front_count() -> tuple[int, str]:
    """返回（前端数, 原因）。原因写进启动日志。"""

        n = max(0, int(env))

    if limit is not None and limit < MIN_MEMORY_FOR_FRONTS:
        return 0, f"内存上限 {limit >> 20} MB < {MIN_MEMORY_FOR_FRONTS >> 20} MB"

## 二、共享状态（flowlens/supervisor.py）

引擎与前端之间通过 multiprocessing 的共享内存交换"降级状态"与"各前端的额度占用"，不需要消息往返。

```python
import multiprocessing as mp

class SharedState:
    """由引擎创建，作为 Process 参数传给前端（spawn 下 sharedctypes 可以这样传递）。
    lock=False：每个槽位只有一个写者（引擎写 level/tight，前端 i 只写自己的第 i 个槽位），
    读者读到的是某一时刻的值即可，不需要锁。"""
    def __init__(self, ctx, n: int):
        self.n = n
        self.level = ctx.Value("i", 0, lock=False)      # 引擎写：当前降级级别
        self.tight = ctx.Value("b", 0, lock=False)      # 引擎写：1 表示第 3 级起收紧额度
        self.pending = ctx.Array("q", n, lock=False)    # 前端 i 写：在途额度已预留的字节
        self.raw_pending = ctx.Array("q", n, lock=False)  # 前端 i 写：原始体额度已预留的字节
        self.ingest_rejected = ctx.Array("q", n, lock=False)  # 前端 i 写：在途额度拒绝累计
        self.raw_rejected = ctx.Array("q", n, lock=False)     # 前端 i 写：原始体额度拒绝累计

    def reset_slot(self, i: int) -> None:
        """前端 i 意外退出后，它的预留不会再释放，由引擎在重启它之前清零。拒绝计数保留。"""
        self.pending[i] = 0
        self.raw_pending[i] = 0
```

前端一侧的额度：继承现有的 Admission（flowlens/ingest.py），每次变化后把值写进自己的槽位。

```python
from .ingest import Admission

class SlotAdmission(Admission):
    """前端本地的额度，同时把占用与拒绝次数镜像到共享数组的第 slot 个槽位。"""
    def __init__(self, limit: int, pending_arr, rejected_arr, slot: int):
        super().__init__(limit=limit)
        self._p, self._r, self._i = pending_arr, rejected_arr, slot

    def try_reserve(self, n: int) -> bool:
        ok = super().try_reserve(n)
        self._p[self._i] = self.pending
        self._r[self._i] = self.rejected
        return ok

    def release(self, n: int) -> None:
        super().release(n)
        self._p[self._i] = self.pending
```

注意 _ingest_expect（flowlens/api/app.py）在拒绝时直接执行 budget.rejected += 1，不经过 try_reserve。前端在每次处理请求结束时（中间件或处理函数的 finally 中）再同步一次 rejected 到共享数组，或者把 rejected 改成属性、在 setter 里同步。二选一，写明选择。

引擎一侧的"视图"：实现现有代码用到的 Admission 接口（Ticker 读 occupancy；_make_degrade_change_hook 设置 limit；_healthz 读 rejected），数据来自共享数组。

```python
class FrontsBudgetView:
    """引擎侧：把各前端的同类额度汇总成一个 Admission 的样子。
    normal_total / tight_total 是全部前端合计的额度；设置 limit 时只翻转共享的 tight 标志，
    实际的每前端额度由前端按 tight 标志自行选择（见第四节）。"""
    def __init__(self, shared: "SharedState", pending_arr, rejected_arr,
                 normal_total: int, tight_total: int):
        self._s, self._p, self._r = shared, pending_arr, rejected_arr
        self._normal, self._tight = normal_total, tight_total

    @property
    def limit(self) -> int:
        return self._tight if self._s.tight.value else self._normal

    @limit.setter
    def limit(self, value: int) -> None:
        self._s.tight.value = 1 if value < self._normal else 0

    @property
    def pending(self) -> int:
        return sum(self._p)

    @property
    def occupancy(self) -> float:
        return self.pending / self.limit if self.limit else 0.0

    @property
    def rejected(self) -> int:
        return sum(self._r)
```

接入方式：前端模式下，make_engine / create_app 用 FrontsBudgetView 替换 ingest_service.admission 与 app[raw_budget_key]。_make_degrade_change_hook 不用改：它执行 admission.limit = tight_adm 时，视图会把共享的 tight 置 1。

降级级别的发布：在 _make_degrade_change_hook 返回的 on_change 里加一行 shared.level.value = new（前端模式下才有 shared，单进程模式下跳过）。前端读取：

## 三、写入通道的消息格式（flowlens/ipc.py，新建）

```python
import asyncio, pickle, struct
from dataclasses import dataclass
import numpy as np

MSG_BATCH = 1   # 前端 → 引擎：一个写入请求的解析结果
MSG_ACK = 2     # 引擎 → 前端：写入结果或错误
_HDR = struct.Struct("<IB")   # 负载长度（字节）+ 消息类型

async def send_msg(writer: asyncio.StreamWriter, mtype: int, obj) -> None:
    data = pickle.dumps(obj, protocol=5)
    writer.write(_HDR.pack(len(data), mtype))
    writer.write(data)
    await writer.drain()

async def read_msg(reader: asyncio.StreamReader):
    """返回 (类型, 对象)。连接关闭时抛 asyncio.IncompleteReadError，由调用方处理。"""
    hdr = await reader.readexactly(_HDR.size)
    n, mtype = _HDR.unpack(hdr)
    return mtype, pickle.loads(await reader.readexactly(n))

@dataclass
class WireBatch:
    """一个写入请求的解析结果的线上格式。字符串只传本批中不重复的那些，再附下标数组。"""
    req_id: int
    ts: np.ndarray       # int64
    nbytes: np.ndarray   # int64
    h_id: np.ndarray     # uint64
    h_url: np.ndarray    # uint64
    h_user: np.ndarray   # uint64
    url_uniq: list
    url_idx: np.ndarray  # int32，长度 = 事件数
    user_uniq: list
    user_idx: np.ndarray
    invalid: int

def _dedupe(seq: list) -> tuple[list, np.ndarray]:
    d: dict = {}
    # setdefault 的第二个参数在插入之前求值，所以新字符串得到的下标就是插入前的 len(d)
    idx = np.fromiter((d.setdefault(s, len(d)) for s in seq), dtype=np.int32, count=len(seq))
    return list(d), idx

def encode_parsed(req_id: int, p) -> WireBatch:
    """p 是 parse_pool.ParsedBatch。"""
    url_uniq, url_idx = _dedupe(p.urls)
    user_uniq, user_idx = _dedupe(p.users)
    return WireBatch(req_id, p.ts, p.nbytes, p.h_id, p.h_url, p.h_user,
                     url_uniq, url_idx, user_uniq, user_idx, p.invalid)

def decode_to_parsed(w: WireBatch, intern_url=None, intern_user=None):
    """引擎侧：只对不重复的字符串做驻留，再按下标还原成逐条的列表。"""
    from .parse_pool import ParsedBatch
    def expand(uniq, idx, intern):
        if intern is not None:
            uniq = [intern(s) for s in uniq]
        arr = np.empty(len(uniq), dtype=object)
        arr[:] = uniq
        return arr[idx].tolist()
    return ParsedBatch(ts=w.ts, nbytes=w.nbytes, h_id=w.h_id, h_url=w.h_url, h_user=w.h_user,
                       urls=expand(w.url_uniq, w.url_idx, intern_url),
                       users=expand(w.user_uniq, w.user_idx, intern_user),
                       invalid=w.invalid)
```

ACK 的负载用元组：成功为 (req_id, "ok", accepted, deduped, late, invalid)；失败为 (req_id, "error", status, message, retry_after)。

驻留：intern_url / intern_user 分别用 flowlens/intern.py 中 url_intern.intern 与 user_intern.intern（为 None 时不驻留）。引擎写入核心 _ingest_core 中现有的 user 驻留（engine.py 第 262 行附近）对已经驻留过的字符串只是一次字典命中，可以保留；如果剖析显示它仍占明显比例，再加一个参数跳过。

测试：
- 分帧：把一条消息的字节拆成多段、分多次喂给 StreamReader，read_msg 能还原
- 编解码：随机生成含大量重复字符串的列表，decode(encode(p)) 后 urls、users 与原始列表逐项相等，数组逐元素相等
- 空批（0 条事件、只有 invalid）也能往返

## 四、前端进程（flowlens/frontend.py，新建）

### 4.1 与引擎的连接

```python
import asyncio, itertools, logging
from .ipc import MSG_BATCH, MSG_ACK, send_msg, read_msg, encode_parsed
from .ingest import IngestResult
from .errors import Overloaded   # 以及其他错误类，用于把引擎的错误还原成同样的异常

class EngineConnection:
    def __init__(self, reader, writer, on_lost):
        self._r, self._w, self._on_lost = reader, writer, on_lost
        self._ids = itertools.count(1)
        self._waiting: dict[int, asyncio.Future] = {}
        self._send_lock = asyncio.Lock()
        self._task = asyncio.create_task(self._read_loop())

    async def submit(self, parsed) -> IngestResult:
        req_id = next(self._ids)
        fut = asyncio.get_running_loop().create_future()
        self._waiting[req_id] = fut
        try:
            async with self._send_lock:            # 一条消息的头和体必须连续写出
                await send_msg(self._w, MSG_BATCH, encode_parsed(req_id, parsed))
            return await fut
        finally:
            self._waiting.pop(req_id, None)

    async def _read_loop(self):
        try:
            while True:
                mtype, msg = await read_msg(self._r)
                if mtype != MSG_ACK:
                    continue
                fut = self._waiting.get(msg[0])
                if fut is None or fut.done():
                    continue
                if msg[1] == "ok":
                    fut.set_result(IngestResult(*msg[2:6]))
                else:
                    fut.set_exception(_error_from_ack(msg))   # 按 status 还原成对应的 FlowLensError
        except (asyncio.IncompleteReadError, ConnectionError):
            pass
        finally:
            for fut in self._waiting.values():
                if not fut.done():
                    fut.set_exception(Overloaded("引擎连接中断，请重试", retry_after=1))
            self._on_lost()      # 通知前端主协程退出
```

### 4.3 每个前端的额度

总额度按前端数平均分配，但每个前端的正常额度不能低于单个请求的上限，否则一个合法的大请求永远预留不到：

```python
def per_front_limit(total: int, n: int, floor: int) -> int:
    return max(total // n, floor)
```

- 在途额度：正常 = per_front_limit(进程池时的在途总额度, n, 解压后单请求上限 4 MB)；收紧 = TIGHT_ADMISSION_LIMIT // n（不设下限，与单进程模式"降级期间单个请求的上限暂时降低"的语义一致）
- 原始体额度：正常 = per_front_limit(RAW_BUDGET_LIMIT, n, MAX_RAW_BODY_BYTES)；收紧 = TIGHT_RAW_BUDGET_LIMIT // n
- 环境变量 FLOWLENS_MAX_PENDING_BYTES、FLOWLENS_RAW_BUDGET_BYTES 仍表示"合计"
- 在 P2-notes.md 中写明：因为有下限，前端数较多时合计会略高于设定值，以及此时的实际合计

### 4.4 前端的 HTTP 应用与转发

复用 flowlens/api/app.py 中的 _error_middleware、_ingest_expect、_read_raw_body（它们只依赖 app[raw_budget_key] 与 app[degrade_controller_key]，前端分别放 SlotAdmission 与 DegradeView）。

```python
from aiohttp import web, ClientSession, UnixConnector, ClientTimeout
from .api.app import _error_middleware, _ingest_expect, _read_raw_body, _json_response, \
    raw_budget_key, degrade_controller_key

front_ingest_key = web.AppKey("front_ingest", FrontIngestService)
upstream_key = web.AppKey("upstream", ClientSession)

_FWD_REQ = {"accept", "last-event-id", "content-type", "content-length", "content-encoding"}
_FWD_RESP = {"content-type", "retry-after", "cache-control", "x-accel-buffering"}

async def _front_ingest(request: web.Request) -> web.Response:
    """与 app.py 的 _ingest 逐行对应，只是写入服务换成 FrontIngestService。"""
    if request.app[degrade_controller_key].reject_ingest:
        raise Overloaded("降级期间暂停写入", retry_after=1)
    body = await _read_raw_body(request, request.app[raw_budget_key])
    r = await request.app[front_ingest_key].ingest(body, request.headers.get("Content-Encoding"))
    return _json_response({"accepted": r.accepted, "deduped": r.deduped,
                           "late": r.late, "invalid": r.invalid}, status=202)

async def _proxy(request: web.Request) -> web.StreamResponse:
    """POST /v1/ingest 以外的一切请求（含不存在的路径、错误的方法）原样交给引擎，
    HTTP 语义只在引擎一处定义。"""
    headers = {k: v for k, v in request.headers.items() if k.lower() in _FWD_REQ}
    data = await request.read() if request.can_read_body else None
    sess = request.app[upstream_key]
    try:
        up_cm = sess.request(request.method, "http://engine" + request.rel_url.path_qs,
                             headers=headers, data=data, allow_redirects=False)
        async with up_cm as up:
            resp = web.StreamResponse(status=up.status, reason=up.reason)
            for k, v in up.headers.items():
                if k.lower() in _FWD_RESP:
                    resp.headers[k] = v
            if up.content_length is not None:
                resp.content_length = up.content_length
            await resp.prepare(request)
            async for chunk in up.content.iter_any():   # SSE：边收边发
                await resp.write(chunk)
            await resp.write_eof()
            return resp
    except (ConnectionError, OSError):
        raise Overloaded("引擎不可用，请重试", retry_after=1)

def create_front_app(conn, shared, slot, n_fronts, query_sock: str) -> web.Application:
    app = web.Application(client_max_size=0, middlewares=[_error_middleware])
    app[raw_budget_key] = SlotAdmission(raw_normal, shared.raw_pending, shared.raw_rejected, slot)
    app[degrade_controller_key] = DegradeView(shared)
    app[front_ingest_key] = FrontIngestService(
        conn, SlotAdmission(adm_normal, shared.pending, shared.ingest_rejected, slot),
        shared, adm_normal, adm_tight)
    app.router.add_post("/v1/ingest", _front_ingest, expect_handler=_ingest_expect)
    app.router.add_route("*", "/{tail:.*}", _proxy)

    async def _open(app):
        app[upstream_key] = ClientSession(connector=UnixConnector(path=query_sock),
                                          auto_decompress=False,
                                          timeout=ClientTimeout(total=None))   # SSE 不能有总超时
    async def _close(app):
        await app[upstream_key].close()
    app.on_startup.append(_open)
    app.on_cleanup.append(_close)
    return app
```

需要验证并在汇报中说明：GET /v1/ingest、PUT /v1/topk 这类"路径存在、方法不对"的请求，在前端被路由到 _proxy、由引擎返回 405（aiohttp 的路由在路径匹配但方法不匹配时会继续尝试后面的资源；如果实测不是这样，改为在前端为 /v1/ingest 显式注册一个返回与引擎一致的 405 的处理器）。

### 4.5 前端进程的入口

```python
def front_main(slot, host, port, ingest_sock, query_sock, shared, n_fronts, lingering):
    """spawn 的目标函数（模块级，可 pickle）。"""
    logging.basicConfig(level=logging.INFO,
                        format=f"%(asctime)s %(levelname)s front{slot} %(name)s %(message)s")
    asyncio.run(_front_run(slot, host, port, ingest_sock, query_sock, shared, n_fronts, lingering))

async def _front_run(slot, host, port, ingest_sock, query_sock, shared, n_fronts, lingering):
    stop = asyncio.Event()
    # 引擎先启动前端、再分配内存和监听，所以这里要重试连接（最多约 30 秒）
    reader = writer = None
    for _ in range(300):
        try:
            reader, writer = await asyncio.open_unix_connection(ingest_sock)
            break
        except (FileNotFoundError, ConnectionRefusedError):
            await asyncio.sleep(0.1)
    if reader is None:
        raise SystemExit(f"front{slot}: 无法连接引擎写入通道 {ingest_sock}")
    conn = EngineConnection(reader, writer, on_lost=stop.set)
    app = create_front_app(conn, shared, slot, n_fronts, query_sock)
    runner = web.AppRunner(app, auto_decompress=False, lingering_time=lingering)
    await runner.setup()
    await web.TCPSite(runner, host, port, reuse_port=True).start()
    loop = asyncio.get_running_loop()
    for sig in (signal.SIGTERM, signal.SIGINT):
        loop.add_signal_handler(sig, stop.set)
    await stop.wait()
    await runner.cleanup()
```

async def serve_front(reader, writer, svc: IngestService):

### 5.4 进程管理（flowlens/supervisor.py）

```python
class FrontSupervisor:
    def __init__(self, n, host, port, ingest_sock, query_sock, lingering):
        self.ctx = mp.get_context("spawn")
        self.shared = SharedState(self.ctx, n)
        self.n, self._args = n, (host, port, ingest_sock, query_sock)
        self._lingering = lingering
        self.procs: list = [None] * n
        self._stopping = False

    def start_all(self):
        for i in range(self.n):
            self._start(i)

    def _start(self, i):
        from .frontend import front_main
        p = self.ctx.Process(target=front_main, name=f"flowlens-front{i}",
                             args=(i, *self._args, self.shared, self.n, self._lingering))
        p.start()
        self.procs[i] = p

    async def monitor(self):
        """每秒检查一次，意外退出的前端：清零它的预留，再重新拉起。"""
        while not self._stopping:
            await asyncio.sleep(1)
            for i, p in enumerate(self.procs):
                if p is not None and not p.is_alive() and not self._stopping:
                    log.warning("前端 %d 退出（exitcode=%s），重新启动", i, p.exitcode)
                    self.shared.reset_slot(i)
                    self._start(i)

    def stop(self, timeout=3.0):
        self._stopping = True
        for p in self.procs:
            if p is not None and p.is_alive():
                p.terminate()                       # SIGTERM
        deadline = time.monotonic() + timeout
        for p in self.procs:
            if p is not None:
                p.join(max(0.0, deadline - time.monotonic()))
                if p.is_alive():
                    p.kill()
                    p.join(1)
```
